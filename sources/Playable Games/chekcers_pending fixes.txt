```js
/* ═══ TUTORIAL: looping auto-played mini game, one canvas ═══
   A scripted real game plays itself on the canvas with full animation and
   wooden sound. Each tutorial page pauses the loop at its teaching beat. */
const TUT_GAME=(()=>{
  /* positions in [row,col,dark,king] form; script of white/black moves */
  const START=[
    [5,0,0,0],[5,2,0,0],[5,4,0,0],[6,3,0,0],
    [1,2,1,0],[1,4,1,0],[2,5,1,0],[0,7,1,0]
  ];
  /* moves as [fromRow,fromCol, toRow,toCol, ...midHops]; dark moves 2nd */
  const SCRIPT=[
    {w:[5,2,4,3]},                       /* white advances */
    {b:[1,4,3,2,2]},                     /* black jumps over (2,3): capture */
    {w:[6,3,4,1,5,2]},                   /* white double-jumps back: 2 caps */
    {b:[2,5,3,4]},                       /* black slips forward */
    {w:[4,1,3,0]},                       /* white men race */
    {b:[3,4,2,5]},
    {w:[3,0,2,1]},
    {b:[2,5,1,6]},
    {w:[2,1,0,3,1,2]},                   /* white jumps, CROWNS on row 0 */
    {b:[1,6,0,7]},                       /* black crowns too */
    {w:[0,3,2,5,1,4]},                   /* white king strikes: capture */
    {b:[0,7,1,6]},
    {w:[2,5,0,7,1,6]}                    /* final shot: black has no moves */
  ];
  let cv=null, ctx=null, anim=null, holdUntil=0, segIdx=0, board=null;
  let raf=0, mutedRef=function(){return true;}, soundRef=null;
  function build(){
    const b=new Int8Array(64);
    for(let i=0;i<START.length;i++){
      const it=START[i];
      b[it[0]*8+it[1]]=it[3]?(it[2]?-2:2):(it[2]?-1:1);
    }
    return b;
  }
  function applySeg(b,seg){
    const path=[];
    const fr=seg.w?seg.w[0]:seg.b[0];
    const fc=seg.w?seg.w[1]:seg.b[1];
    path.push(fr*8+fc);
    const steps=seg.w?seg.w:seg.b;
    for(let i=2;i<steps.length;i+=2){
      path.push(steps[i]*8+steps[i+1]);
    }
    const dark=seg.b?true:false;
    let val=b[path[0]];
    b[path[0]]=0;
    let cur=path[0];
    const caps=[];
    for(let i=1;i<path.length;i++){
      const s=path[i];
      if(Math.abs((s>>3)-(cur>>3))===2){
        const mid=(cur+s)>>1;
        caps.push({sq:mid,val:b[mid]});
        b[mid]=0;
      }
      cur=s;
    }
    b[cur]=val;
    if(Math.abs(val)===1&&(cur>>3)===(dark?7:0)) b[cur]=dark?-2:2;
    return {path:path,caps:caps,dark:dark,val:val,crowned:(Math.abs(val)===1)};
  }
  function draw(){
    if(!ctx) return;
    const d=Math.min(window.devicePixelRatio||1,2);
    const css=Math.min(300,(document.getElementById("app").clientWidth||340)-40);
    const px=Math.round(css*d);
    cv.width=px; cv.height=px;
    cv.style.width=css+"px"; cv.style.height=css+"px";
    ctx.setTransform(d,0,0,d,0,0);
    const cs=CFG.colors;
    const m=Math.max(4,Math.round(css*0.02));
    const q=(css-2*m)/8;
    ctx.fillStyle=cs.frame; ctx.fillRect(0,0,css,css);
    for(let r=0;r<8;r++)for(let c=0;c<8;c++){
      ctx.fillStyle=((r+c)&1)===0?cs.lightSq:cs.darkSq;
      ctx.fillRect(m+c*q,m+r*q,q+0.5,q+0.5);
    }
    const cx=col=>m+col*q+q/2, cy=row=>m+row*q+q/2;
    let moving=null;
    if(anim){
      const segs=anim.path.length-1;
      const per=anim.dur/segs;
      let t=anim.t, si=0;
      while(si<segs-1&&t>per){ t-=per; si++; }
      const k=easeOutCubic(Math.min(1,t/per));
      for(let i=0;i<anim.caps.length;i++){
        if(i>=si+1) continue;
        const mid=anim.caps[i];
        ctx.globalAlpha=Math.max(0.15,1-k);
        drawD(ctx,cx(mid.sq%8),cy(mid.sq>>3),q*0.4,mid.val<0,Math.abs(mid.val)===2);
        ctx.globalAlpha=1;
      }
      const x=cx(anim.path[si]%8)+(cx(anim.path[si+1]%8)-cx(anim.path[si]%8))*k;
      const y=cy(anim.path[si]>>3)+(cy(anim.path[si+1]>>3)-cy(anim.path[si]>>3))*k;
      moving={x:x,y:y,dark:anim.dark,king:anim.crowned||(Math.abs(anim.val)===2)};
    }
    for(let s=0;s<64;s++){
      const v=board[s]; if(v===0) continue;
      if(anim&&(s===anim.path[0]||anim.caps.some(function(cp){return cp.sq===s;}))) continue;
      drawD(ctx,cx(s%8),cy(s>>3),q*0.4,v<0,Math.abs(v)===2);
    }
    if(moving) drawD(ctx,moving.x,moving.y,q*0.42,moving.dark,moving.king);
  }
  function drawD(c,x,y,rad,dark,king){
    const cs=CFG.colors;
    const F=dark?cs.bFill:cs.wFill, S=dark?cs.bStroke:cs.wStroke;
    c.beginPath(); c.arc(x,y,rad,0,Math.PI*2);
    c.fillStyle=F; c.fill();
    c.lineWidth=Math.max(2,rad*0.12); c.strokeStyle=S; c.stroke();
    c.beginPath(); c.arc(x,y,rad*0.68,0,Math.PI*2);
    c.lineWidth=Math.max(1,rad*0.07); c.strokeStyle=cs.groove; c.stroke();
    if(king){
      c.beginPath(); c.arc(x,y,rad*0.86,0,Math.PI*2);
      c.lineWidth=Math.max(1.5,rad*0.09); c.strokeStyle=cs.kingGold; c.stroke();
      const w=rad*0.62, h=rad*0.4;
      c.beginPath();
      c.moveTo(x-w/2,y+h/2);
      c.lineTo(x-w/2,y-h/4);
      c.lineTo(x-w/4,y+h/8);
      c.lineTo(x,y-h/2);
      c.lineTo(x+w/4,y+h/8);
      c.lineTo(x+w/2,y-h/4);
      c.lineTo(x+w/2,y+h/2);
      c.closePath();
      c.fillStyle=cs.kingGold; c.fill();
    }
  }
  function step(ts){
    raf=requestAnimationFrame(step);
    if(!cv||cv.style.display==="none") return;
    if(!anim){
      if(holdUntil>0){ /* page hold: game breathes but does not advance */ }
      else if(segIdx<SCRIPT.length){
        const seg=SCRIPT[segIdx];
        const view=applySeg(board,seg);
        segIdx++;
        anim={t:0,dur:200*(view.path.length-1),
          path:view.path,caps:view.caps,dark:view.dark,val:view.val,
          crowned:view.crowned};
        if(soundRef&&!mutedRef()){
          const hops=view.path.length-1;
          for(let h=1;h<=hops;h++){
            (function(hh){ setTimeout(function(){ soundRef(hh-1); },hh*200); })(h);
          }
        }
      } else {
        holdUntil=Date.now()+2000;   /* end-of-game hold, then loop */
      }
    } else {
      anim.t+=16;
      if(anim.t>=anim.dur){ anim=null; }
    }
    if(holdUntil>0&&Date.now()>holdUntil){
      holdUntil=0; segIdx=0; board=build();
    }
    draw();
  }
  return {
    start:function(canvas,playSound,isMuted){
      cv=canvas; soundRef=playSound; mutedRef=isMuted;
      if(!board) board=build();
      segIdx=0; holdUntil=0; anim=null;
      if(!raf) raf=requestAnimationFrame(step);
    },
    /* page focus: hold the loop at the beat matching the lesson */
    focus:function(page){
      /* beat indices chosen per page: 0=start 1=capture 3=double 8=crowning 12=end */
      const beats=[0,1,2,8,12];
      segIdx=beats[Math.max(0,Math.min(4,page))];
      holdUntil=0; anim=null; board=build();
      /* pre-apply the board up to the beat so the frame matches the story */
      for(let i=0;i<segIdx;i++) applySeg(board,SCRIPT[i]);
    },
    stop:function(){ if(raf){ cancelAnimationFrame(raf); raf=0; } }
  };
})();
```
